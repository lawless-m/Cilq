H='Authorization: Bearer BRIDGE'; C='Content-Type: application/json'
curl -s localhost:3141/workers -H "$H"
curl -s -XPOST localhost:3141/jobs/sync -H "$H" -H "$C" -d '{"script":"bridge.nodes()"}'
curl -s -XPOST localhost:3141/jobs/sync -H "$H" -H "$C" -d '{"script":"bridge.node(\"cart-total\").textContent"}'
